18: def extract_robust_features(
19:     windows: np.ndarray, sfreq: float, chunk_size: int = 256
20: ) -> np.ndarray:
21:     """Extract 10 features/channel in bounded-memory chunks.
22: 
23:     Input is (n_windows, n_channels, n_samples); output is finite float32.
24:     Chunking limits temporary float64 and Welch PSD allocations without
25:     changing feature definitions or the order of output rows/features.
26:     """
27:     values = np.asarray(windows)
28:     if values.ndim != 3 or any(size == 0 for size in values.shape):
29:         raise ValueError("Expected windows shaped (n_windows, n_channels, n_samples).")
30:     if not isinstance(chunk_size, (int, np.integer)) or isinstance(chunk_size, bool) or chunk_size <= 0:
31:         raise ValueError("chunk_size must be a positive integer.")
32:     if not np.isscalar(sfreq) or not np.isfinite(sfreq) or sfreq < 80:
33:         raise ValueError("Sampling frequency must be finite and at least 80 Hz.")
34:     minimum_samples = int(np.ceil(2.0 * float(sfreq)))
35:     if values.shape[-1] < minimum_samples:
36:         raise ValueError(
37:             f"Each window must contain at least 2 seconds of EEG "
38:             f"({minimum_samples} samples at {sfreq} Hz)."
39:         )
40: 
41:     n_windows, n_channels, _ = values.shape
42:     output = np.empty(
43:         (n_windows, n_channels, len(ROBUST_FEATURE_NAMES)), dtype=np.float32
44:     )
45:     eps = np.finfo(np.float64).eps
46:     nperseg = min(values.shape[-1], max(1, int(round(2.0 * float(sfreq)))))
47: 
48:     for start in range(0, n_windows, int(chunk_size)):
49:         stop = min(start + int(chunk_size), n_windows)
50:         chunk = np.asarray(values[start:stop], dtype=np.float64)
51:         if not np.isfinite(chunk).all():
52:             raise ValueError("EEG windows contain NaN or infinite values.")
53: 
54:         centered = chunk - np.mean(chunk, axis=-1, keepdims=True)
55:         std = np.std(centered, axis=-1)
56:         safe_std = np.maximum(std, eps)
57:         log_std = np.log(np.maximum(std, 1e-12))
58:         normalized_line_length = (
59:             np.mean(np.abs(np.diff(chunk, axis=-1)), axis=-1) / safe_std
60:         )
61:         skewness = np.mean(centered ** 3, axis=-1) / (safe_std ** 3)
62:         kurtosis = np.mean(centered ** 4, axis=-1) / (safe_std ** 4) - 3.0
63:         near_constant = std <= eps * np.maximum(
64:             1.0, np.max(np.abs(chunk), axis=-1)
65:         )
66:         skewness = np.where(near_constant, 0.0, skewness)
67:         kurtosis = np.where(near_constant, 0.0, kurtosis)
68: 
69:         frequencies, psd = welch(
70:             chunk, fs=float(sfreq), nperseg=nperseg, axis=-1,
71:             detrend="constant", scaling="density",
72:         )
73:         df = float(frequencies[1] - frequencies[0])
74:         band_powers = []
75:         for low, high in ROBUST_FREQUENCY_BANDS:
76:             mask = (frequencies >= low) & (frequencies < high)
77:             band_powers.append(np.sum(psd[..., mask], axis=-1) * df)
78:         powers = np.stack(band_powers, axis=-1)
79:         total_power = np.sum(powers, axis=-1, keepdims=True)
80:         relative = np.divide(
81:             powers, total_power, out=np.zeros_like(powers), where=total_power > eps
82:         )
83: 
84:         retained = (frequencies >= 0.5) & (frequencies <= 40.0)
85:         spectral = np.maximum(psd[..., retained], 0.0)
86:         spectral_total = np.sum(spectral, axis=-1, keepdims=True)
87:         probabilities = np.divide(
88:             spectral, spectral_total, out=np.zeros_like(spectral),
89:             where=spectral_total > eps,
90:         )
91:         log_probabilities = np.zeros_like(probabilities)
92:         positive = probabilities > 0
93:         log_probabilities[positive] = np.log(probabilities[positive])
94:         entropy = -np.sum(
95:             probabilities * log_probabilities, axis=-1
96:         ) / np.log(max(2, spectral.shape[-1]))
97: 
98:         output[start:stop] = np.stack([
99:             log_std, normalized_line_length, skewness, kurtosis,
100:             *[relative[..., index] for index in range(relative.shape[-1])],
101:             entropy,
102:         ], axis=-1).astype(np.float32)
103: 
104:     result = output.reshape(n_windows, -1)
105:     if not np.isfinite(result).all():
106:         raise ValueError("Robust feature extraction generated non-finite values.")
107:     return result

86:     def partial_fit(self, training_windows):
87:         """Merge one recording/batch into the running population statistics."""
88:         if self.mean_ is not None:
89:             raise RuntimeError("Cannot partial_fit after the scaler has been finalized.")
90:         values = self._validate_windows(training_windows)
91:         if self._n_channels is None:
92:             self._n_channels = values.shape[1]
93:         elif values.shape[1] != self._n_channels:
94:             raise ValueError("Training batches have inconsistent channel counts.")
95: 
96:         flattened = values.transpose(1, 0, 2).reshape(values.shape[1], -1)
97:         batch_count = flattened.shape[1]
98:         batch_mean = np.mean(flattened, axis=1)
99:         centered = flattened - batch_mean[:, None]
100:         batch_m2 = np.sum(centered * centered, axis=1)
101: 
102:         if self._count == 0:
103:             self._count = int(batch_count)
104:             self._mean = batch_mean
105:             self._m2 = batch_m2
106:         else:
107:             total = self._count + batch_count
108:             delta = batch_mean - self._mean
109:             self._mean = self._mean + delta * (batch_count / total)
110:             self._m2 = (
111:                 self._m2 + batch_m2
112:                 + delta * delta * (self._count * batch_count / total)
113:             )
114:             self._count = int(total)
115:         return self

118:         """Return a detached, serializable snapshot of the running accumulator."""
119:         return {
120:             "count": int(self._count),
121:             "n_channels": int(self._n_channels) if self._n_channels is not None else 0,
122:             "mean": None if self._mean is None else self._mean.copy(),
123:             "m2": None if self._m2 is None else self._m2.copy(),
124:         }

126:     def load_state_dict(self, state):
127:         """Restore and validate an unfinished accumulator checkpoint."""
128:         if self.mean_ is not None:
129:             raise RuntimeError("Cannot restore state after the scaler has been finalized.")
130:         try:
131:             count = int(state["count"])
132:             n_channels = int(state["n_channels"])
133:             mean = np.asarray(state["mean"], dtype=np.float64)
134:             m2 = np.asarray(state["m2"], dtype=np.float64)
135:         except (KeyError, TypeError, ValueError) as exc:
136:             raise ValueError("Invalid scaler accumulator checkpoint.") from exc
137:         if count <= 0 or n_channels <= 0:
138:             raise ValueError("Scaler checkpoint must contain completed samples and channels.")
139:         if mean.shape != (n_channels,) or m2.shape != (n_channels,):
140:             raise ValueError("Scaler checkpoint arrays have inconsistent dimensions.")
141:         if not np.isfinite(mean).all() or not np.isfinite(m2).all() or np.any(m2 < 0):
142:             raise ValueError("Scaler checkpoint contains invalid statistics.")
143:         self._count = count
144:         self._n_channels = n_channels
145:         self._mean = mean.copy()
146:         self._m2 = m2.copy()

149:     def finalize(self):
150:         """Convert running accumulators into immutable transform statistics."""
151:         if self._count == 0 or self._mean is None or self._m2 is None:
152:             raise ValueError("Cannot fit standardizer on no training windows.")
153:         variance = np.maximum(self._m2 / self._count, 0.0)
154:         self.n_channels_ = int(self._n_channels)
155:         self.n_values_per_channel_ = int(self._count)
156:         self.mean_ = self._mean.copy()
157:         self.scale_ = np.maximum(np.sqrt(variance), self.epsilon)
158:         return self

161:         return self.fit_batches([training_windows])
