56:     def iter_split(self, split: str) -> Iterator[RecordingWindows]:
57:         """Stream fold-standardized EEG windows, one recording at a time."""
58:         groups = {
59:             "fit": self.fit_recordings,
60:             "validation": self.validation_recordings,
61:             "test": self.test_recordings,
62:         }
63:         if split not in groups:
64:             raise ValueError("split must be one of: fit, validation, test")
65:         for recording in groups[split]:
66:             item = load_recording_windows(self.processed_dir, recording)
67:             yield RecordingWindows(
68:                 recording, self.standardizer.transform(item.X),
69:                 item.y, item.starts, item.sfreq,
70:             )
71: 
72:     def iter_features(self, split: str) -> Iterator[RecordingFeatures]:
73:         """Stream deterministic feature matrices, one recording at a time.
74: 
75:         Channel standardization is applied by iter_split before extraction.
76:         The extractor itself does not fit on labels or data, so validation
77:         and outer-test recordings cannot influence fitted preprocessing.
78:         """
79:         expected_features = len(PREFERRED_CHANNELS) * len(FEATURE_NAMES)
80:         for item in self.iter_split(split):
81:             matrix = extract_features(item.X, item.sfreq)
82:             if matrix.shape != (len(item.y), expected_features):
83:                 raise RuntimeError(
84:                     "Feature extractor returned an unexpected shape: "
85:                     f"{matrix.shape}; expected ({len(item.y)}, {expected_features})."
86:                 )
87:             if matrix.dtype != np.float32 or not np.isfinite(matrix).all():
88:                 raise RuntimeError("Feature matrix must be finite float32.")
89:             yield RecordingFeatures(
90:                 item.recording, matrix, item.y, item.starts, item.sfreq
91:             )

98: def load_recording_windows(processed_dir: str | Path, recording: Recording) -> RecordingWindows:
99:     path = processed_path_for_recording(processed_dir, recording)
100:     if not path.is_file():
101:         raise FileNotFoundError(
102:             f"Processed windows missing for {recording.relative_path}: {path}"
103:         )
104:     with np.load(path, allow_pickle=False) as saved:
105:         missing = _REQUIRED_ARRAYS - set(saved.files)
106:         if missing:
107:             raise ValueError(f"{path.name} is missing arrays: {', '.join(sorted(missing))}")
108:         X = np.asarray(saved["X"])
109:         y = np.asarray(saved["y"])
110:         starts = np.asarray(saved["starts"], dtype=np.float64)
111:         channels = [str(value) for value in saved["channels"].tolist()]
112:         sfreq = float(saved["sfreq"])
113:         subject = str(saved["subject"])
114:         source_file = str(saved["source_file"])
115:         normalization = str(saved["normalization"])
116: 
117:     if subject != recording.subject or source_file != recording.edf_file:
118:         raise ValueError(f"Saved recording identity does not match manifest: {path.name}")
119:     if normalization != _NORMALIZATION_POLICY:
120:         raise ValueError(
121:             f"{path.name} uses incompatible normalization policy; regenerate windows."
122:         )
123:     if channels != list(PREFERRED_CHANNELS):
124:         raise ValueError(f"{path.name} does not use the configured fixed channel order.")
125:     if X.ndim != 3 or X.shape[1] != len(PREFERRED_CHANNELS):
126:         raise ValueError(f"{path.name} has invalid X shape {X.shape}.")
127:     if y.ndim != 1 or starts.ndim != 1 or len(X) != len(y) or len(y) != len(starts):
128:         raise ValueError(f"{path.name} has inconsistent X/y/starts lengths.")
129:     if len(y) == 0 or not np.isin(y, [0, 1]).all():
130:         raise ValueError(f"{path.name} has empty or non-binary labels.")
131:     if not np.isfinite(X).all() or not np.isfinite(starts).all() or not np.isfinite(sfreq):
132:         raise ValueError(f"{path.name} contains non-finite values.")
133:     if sfreq <= 0:
134:         raise ValueError(f"{path.name} has invalid sampling frequency.")
135:     if bool(np.any(y == 1)) != (recording.seizure_count > 0):
136:         raise ValueError(
137:             f"{path.name} labels disagree with audited seizure annotation presence."
138:         )
139:     return RecordingWindows(recording, X, y.astype(np.int64), starts, sfreq)

142: def prepare_lopo_fold(
143:     fold: LopoFold,
144:     processed_dir: str | Path,
145:     validation_fraction: float = 0.2,
146:     random_state: int = 42,
147:     scaler_checkpoint_dir: str | Path | None = None,
148: ) -> PreparedFold:
149:     """Split outer-training patients, then fit scaler on inner-fit recordings only."""
150:     train_patients = {item.patient_id for item in fold.train_recordings}
151:     test_patients = {item.patient_id for item in fold.test_recordings}
152:     if test_patients != {fold.test_patient_id}:
153:         raise ValueError("Outer fold test recordings do not match its test patient.")
154:     if train_patients & test_patients:
155:         raise ValueError("Patient leakage detected in supplied outer fold.")
156: 
157:     inner = build_patient_validation_split(
158:         fold.train_recordings, validation_fraction, random_state
159:     )
160:     fit_patients = set(inner.train_patient_ids)
161:     validation_patients = set(inner.validation_patient_ids)
162:     if fit_patients & validation_patients or fit_patients & test_patients:
163:         raise RuntimeError("Patient leakage detected before normalization fitting.")
164:     if validation_patients & test_patients:
165:         raise RuntimeError("Outer test patient entered inner validation.")
166: 
167:     directory = Path(processed_dir)
168:     scaler = FoldChannelStandardizer()
169: 
170:     def training_batches(start_index=0):
171:         for recording in inner.train_recordings[start_index:]:
172:             yield load_recording_windows(directory, recording).X
173: 
174:     if scaler_checkpoint_dir is None:
175:         scaler.fit_batches(training_batches())
176:     else:
177:         checkpoint_context = {
178:             "stage": "inner_fit_channel_scaler",
179:             "fold_id": fold.fold_id,
180:             "outer_test_patient": fold.test_patient_id,
181:             "inner_fit_patients": sorted(fit_patients),
182:             "inner_validation_patients": sorted(validation_patients),
183:             "random_state": int(random_state),
184:             "validation_fraction": float(validation_fraction),
185:         }
186:         signature = scaler_checkpoint_signature(
187:             inner.train_recordings, directory, checkpoint_context, scaler.epsilon
188:         )
189:         scaler.fit_batches_checkpointed(
190:             training_batches,
191:             total_batches=len(inner.train_recordings),
192:             checkpoint_dir=scaler_checkpoint_dir,
193:             signature=signature,
194:             progress_label="inner-fit channel-scaler fit",
195:         )
196:     return PreparedFold(
197:         fold.fold_id, fold.test_patient_id,
198:         inner.train_recordings, inner.validation_recordings,
199:         fold.test_recordings, scaler, directory,
200:     )
