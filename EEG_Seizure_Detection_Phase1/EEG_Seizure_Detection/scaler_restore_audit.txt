78:         if values.ndim != 3:

86:     def partial_fit(self, training_windows):
87:         """Merge one recording/batch into the running population statistics."""
88:         if self.mean_ is not None:
89:             raise RuntimeError("Cannot partial_fit after the scaler has been finalized.")
90:         values = self._validate_windows(training_windows)
91:         if self._n_channels is None:
92:             self._n_channels = values.shape[1]
93:         elif values.shape[1] != self._n_channels:
94:             raise ValueError("Training batches have inconsistent channel counts.")
95: 
96:         flattened = values.transpose(1, 0, 2).reshape(values.shape[1], -1)
97:         batch_count = flattened.shape[1]
98:         batch_mean = np.mean(flattened, axis=1)
99:         centered = flattened - batch_mean[:, None]
100:         batch_m2 = np.sum(centered * centered, axis=1)
101: 
102:         if self._count == 0:
103:             self._count = int(batch_count)
104:             self._mean = batch_mean
105:             self._m2 = batch_m2
106:         else:
107:             total = self._count + batch_count
108:             delta = batch_mean - self._mean
109:             self._mean = self._mean + delta * (batch_count / total)
110:             self._m2 = (
111:                 self._m2 + batch_m2
112:                 + delta * delta * (self._count * batch_count / total)
113:             )
114:             self._count = int(total)
115:         return self

117:     def state_dict(self):
118:         """Return a detached, serializable snapshot of the running accumulator."""
119:         return {
120:             "count": int(self._count),
121:             "n_channels": int(self._n_channels) if self._n_channels is not None else 0,
122:             "mean": None if self._mean is None else self._mean.copy(),
123:             "m2": None if self._m2 is None else self._m2.copy(),
124:         }

126:     def load_state_dict(self, state):
127:         """Restore and validate an unfinished accumulator checkpoint."""
128:         if self.mean_ is not None:
129:             raise RuntimeError("Cannot restore state after the scaler has been finalized.")
130:         try:
131:             count = int(state["count"])
132:             n_channels = int(state["n_channels"])
133:             mean = np.asarray(state["mean"], dtype=np.float64)
134:             m2 = np.asarray(state["m2"], dtype=np.float64)
135:         except (KeyError, TypeError, ValueError) as exc:
136:             raise ValueError("Invalid scaler accumulator checkpoint.") from exc
137:         if count <= 0 or n_channels <= 0:
138:             raise ValueError("Scaler checkpoint must contain completed samples and channels.")
139:         if mean.shape != (n_channels,) or m2.shape != (n_channels,):
140:             raise ValueError("Scaler checkpoint arrays have inconsistent dimensions.")
141:         if not np.isfinite(mean).all() or not np.isfinite(m2).all() or np.any(m2 < 0):
142:             raise ValueError("Scaler checkpoint contains invalid statistics.")
143:         self._count = count
144:         self._n_channels = n_channels
145:         self._mean = mean.copy()
146:         self._m2 = m2.copy()
147:         return self

149:     def finalize(self):
150:         """Convert running accumulators into immutable transform statistics."""
151:         if self._count == 0 or self._mean is None or self._m2 is None:
152:             raise ValueError("Cannot fit standardizer on no training windows.")
153:         variance = np.maximum(self._m2 / self._count, 0.0)
154:         self.n_channels_ = int(self._n_channels)
155:         self.n_values_per_channel_ = int(self._count)
156:         self.mean_ = self._mean.copy()
157:         self.scale_ = np.maximum(np.sqrt(variance), self.epsilon)
158:         return self
